        partial void SelfTestTick()
        {
            try
            {
                SelfTestTickCore();
            }
            catch (System.Exception e)
            {
                if (m_SelfTestReport != null)
                {
                    m_SelfTestReport.AppendLine($"FAIL | izuzetak u fazi {m_SelfTestPhase}: {e.Message}");
                    SelfTestFinish();
                }

                m_SelfTestPhase = 0;
            }
        }

        private void SelfTestTickCore()
        {
            SelfTestFollowUp();
            SelfTestRecheck();

            if (m_SelfTestPhase == 0)
            {
                UnityEngine.InputSystem.Keyboard keyboard = UnityEngine.InputSystem.Keyboard.current;
                if (keyboard == null || m_Mode != Mode.Select ||
                    !keyboard.ctrlKey.isPressed || !keyboard.shiftKey.isPressed)
                {
                    return;
                }

                // F10: sonda nad STVARNOM selekcijom u gradu (bez pravljenja
                // testnog puta) — dopisuje snimak u probe.txt.
                if (keyboard.f10Key.wasPressedThisFrame)
                {
                    SelfTestProbeSelection();
                    return;
                }

                if (m_HandleDragging || !keyboard.f9Key.wasPressedThisFrame)
                {
                    return;
                }

                SelfTestStart();
                return;
            }

            if (m_SelfTestWait > 0)
            {
                m_SelfTestWait--;
                return;
            }

            switch (m_SelfTestPhase)
            {
                case 1: SelfTestPhaseFind(); break;
                case 2: SelfTestPhaseDelete(); break;
                case 3: SelfTestPhaseAssertDeleteAndUndo(); break;
                case 4: SelfTestPhaseAssertUndo(); break;
                case 5: SelfTestPhaseStraighten(); break;
                case 6: SelfTestPhaseAssertStraighten(); break;
                case 7: SelfTestPhaseFilterAndHistory(); break;
                case 8: SelfTestPhaseTrafficBaseline(); break;
                case 9: SelfTestPhaseTrafficMoveNode(); break;
                case 10: SelfTestPhaseTrafficAfterMove(); break;
                case 11: SelfTestPhaseTrafficHandleCurrent(); break;
                case 12: SelfTestPhaseTrafficAfterCurrent(); break;
                case 13: SelfTestPhaseTrafficHandleFixed(); break;
                case 14: SelfTestPhaseTrafficAfterFixed(); break;
                case 15: SelfTestPhaseCleanup(); break;
                case 16: SelfTestPhaseVerifyCleanup(); break;
                default: m_SelfTestPhase = 0; break;
            }
        }

        private void SelfTestStart()
        {
            if (!GetRaycastResult(out Entity _, out Game.Common.RaycastHit hit))
            {
                Mod.Log.Info("Copaste self-test: uperi kursor u prazan teren pa Ctrl+Shift+F9");
                return;
            }

            m_SelfTestPrefab = Entity.Null;
            foreach (PrefabID candidate in kSelfTestPrefabs)
            {
                if (m_PrefabSystem.TryGetPrefab(candidate, out PrefabBase prefabBase) && prefabBase != null)
                {
                    m_SelfTestPrefab = m_PrefabSystem.GetEntity(prefabBase);
                    break;
                }
            }

            if (m_SelfTestPrefab == Entity.Null)
            {
                Mod.Log.Warn("Copaste self-test: nijedan probni prefab nije nađen");
                return;
            }

            m_SelfTestReport = new StringBuilder();
            m_SelfTestReport.AppendLine("=== Copaste self-test ===");
            m_SelfTestFailures = 0;
            m_SelfTestNetworksWas = Mod.Settings != null && Mod.Settings.SelectNetworks;
            if (Mod.Settings != null)
            {
                Mod.Settings.SelectNetworks = true;
            }

            ClearSelection();
            m_SelfTestUndoBackup = new List<UndoRecord>(m_UndoStack);
            m_SelfTestRedoBackup = new List<UndoRecord>(m_RedoStack);

            // Kraki put: A---B---C, B namerno 12 m VAN prave A–C (za ravnanje).
            TerrainHeightData heightData = m_TerrainSystem.GetHeightData();
            float3 origin = hit.m_HitPosition;
            m_SelfTestA = new float3(origin.x - 30f, 0f, origin.z);
            m_SelfTestB = new float3(origin.x, 0f, origin.z + 12f);
            m_SelfTestC = new float3(origin.x + 30f, 0f, origin.z);
            m_SelfTestA.y = TerrainUtils.SampleHeight(ref heightData, m_SelfTestA);
            m_SelfTestB.y = TerrainUtils.SampleHeight(ref heightData, m_SelfTestB);
            m_SelfTestC.y = TerrainUtils.SampleHeight(ref heightData, m_SelfTestC);

            List<NetEdgeSnapshot> snapshots = new List<NetEdgeSnapshot>
            {
                SelfTestSnapshot(m_SelfTestA, m_SelfTestB),
                SelfTestSnapshot(m_SelfTestB, m_SelfTestC),
            };

            RecreateNetEdges(snapshots);
            Mod.Log.Info("Copaste self-test: start");
            m_SelfTestPhase = 1;
            m_SelfTestWait = 10;
        }

        private NetEdgeSnapshot SelfTestSnapshot(float3 from, float3 to)
        {
            Bezier4x3 curve = new Bezier4x3(from, math.lerp(from, to, 1f / 3f), math.lerp(from, to, 2f / 3f), to);
            return new NetEdgeSnapshot
            {
                m_Entity = Entity.Null,
                m_Curve = curve,
                m_Prefab = m_SelfTestPrefab,
                m_StartNode = Entity.Null,
                m_EndNode = Entity.Null,
                m_HasNodePositions = true,
                m_StartNodePos = from,
                m_EndNodePos = to,
            };
        }

        private void SelfTestCheck(bool ok, string name, string detail)
        {
            if (!ok)
            {
                m_SelfTestFailures++;
            }

            string suffix = string.IsNullOrEmpty(detail) ? string.Empty : " | " + detail;
            m_SelfTestReport.AppendLine((ok ? "PASS" : "FAIL") + " | " + name + suffix);
        }

        // Sve probne ivice u krugu od 80 m oko sredine testa.
        private List<Entity> SelfTestScan()
        {
            List<Entity> found = new List<Entity>();
            if (m_NetSearchSystem == null)
            {
                m_NetSearchSystem = World.GetOrCreateSystemManaged<Game.Net.SearchSystem>();
            }

            NativeQuadTree<Entity, QuadTreeBoundsXZ> tree = m_NetSearchSystem.GetNetSearchTree(true, out JobHandle dependencies);
            dependencies.Complete();

            float3 center = (m_SelfTestA + m_SelfTestC) * 0.5f;
            RoadIterator iterator = new RoadIterator
            {
                m_Bounds = new Bounds3(center - new float3(80f, 1000f, 80f), center + new float3(80f, 1000f, 80f)),
                m_Results = new NativeList<Entity>(16, Allocator.Temp),
            };
            tree.Iterate(ref iterator, 0);

            for (int i = 0; i < iterator.m_Results.Length; i++)
            {
                Entity candidate = iterator.m_Results[i];
                if (EntityManager.HasComponent<Game.Net.Edge>(candidate) &&
                    !EntityManager.HasComponent<Owner>(candidate) &&
                    !EntityManager.HasComponent<Game.Tools.Temp>(candidate) &&
                    !EntityManager.HasComponent<Deleted>(candidate) &&
                    EntityManager.TryGetComponent(candidate, out PrefabRef prefabRef) &&
                    prefabRef.m_Prefab == m_SelfTestPrefab)
                {
                    found.Add(candidate);
                }
            }

            iterator.m_Results.Dispose();
            return found;
        }

        // Zajednički čvor dve ivice na tački B — srce testa šava.
        private Entity SelfTestSharedNode(List<Entity> edges)
        {
            Dictionary<Entity, int> counts = new Dictionary<Entity, int>();
            foreach (Entity edge in edges)
            {
                if (!EntityManager.TryGetComponent(edge, out Game.Net.Edge edgeData))
                {
                    continue;
                }

                foreach (Entity node in new[] { edgeData.m_Start, edgeData.m_End })
                {
                    if (EntityManager.TryGetComponent(node, out Game.Net.Node nodeData) &&
                        math.distance(nodeData.m_Position.xz, m_SelfTestB.xz) < 3f)
                    {
                        counts.TryGetValue(node, out int count);
                        counts[node] = count + 1;
                    }
                }
            }

            foreach (KeyValuePair<Entity, int> pair in counts)
            {
                if (pair.Value >= 2)
                {
                    return pair.Key;
                }
            }

            return Entity.Null;
        }

        private void SelfTestPhaseFind()
        {
            List<Entity> edges = SelfTestScan();
            SelfTestCheck(edges.Count == 2, "PRAVLJENJE: dve deonice nastale", "nađeno " + edges.Count);
            m_SelfTestNodeB = SelfTestSharedNode(edges);
            SelfTestCheck(m_SelfTestNodeB != Entity.Null, "ŠAV: deonice dele čvor B", string.Empty);
            if (edges.Count != 2 || m_SelfTestNodeB == Entity.Null)
            {
                m_SelfTestPhase = 15;
                m_SelfTestWait = 2;
                return;
            }

            // Ivica 1 = ona čija je sredina bliža sredini A–B.
            float3 midAB = (m_SelfTestA + m_SelfTestB) * 0.5f;
            EntityManager.TryGetComponent(edges[0], out Game.Net.Curve curve0);
            float d0 = math.distance(LaneMidpoint(curve0.m_Bezier).xz, midAB.xz);
            EntityManager.TryGetComponent(edges[1], out Game.Net.Curve curve1);
            float d1 = math.distance(LaneMidpoint(curve1.m_Bezier).xz, midAB.xz);
            m_SelfTestEdge1 = d0 <= d1 ? edges[0] : edges[1];
            m_SelfTestEdge2 = d0 <= d1 ? edges[1] : edges[0];
            m_SelfTestPhase = 2;
            m_SelfTestWait = 2;
        }

        private void SelfTestPhaseDelete()

        private void SelfTestPhaseAssertDeleteAndUndo()
        {
            bool edge1Gone = !EntityManager.Exists(m_SelfTestEdge1) || EntityManager.HasComponent<Deleted>(m_SelfTestEdge1);
            bool edge2Alive = EntityManager.Exists(m_SelfTestEdge2) && !EntityManager.HasComponent<Deleted>(m_SelfTestEdge2);
            SelfTestCheck(edge1Gone, "BRISANJE: obrisana deonica nestala", string.Empty);
            SelfTestCheck(edge2Alive, "BRISANJE: susedna deonica netaknuta", string.Empty);
            Undo();
            m_SelfTestPhase = 4;
            m_SelfTestWait = 14;
        }

        private void SelfTestPhaseAssertUndo()
        {
            List<Entity> edges = SelfTestScan();
            SelfTestCheck(edges.Count == 2, "UNDO BRISANJA: deonica se vratila", "nađeno " + edges.Count);
            m_SelfTestNodeB = SelfTestSharedNode(edges);
            SelfTestCheck(m_SelfTestNodeB != Entity.Null, "UNDO BRISANJA: šav ponovo ceo (dele čvor)", string.Empty);
            if (edges.Count == 2 && m_SelfTestNodeB != Entity.Null)
            {
                float3 midAB = (m_SelfTestA + m_SelfTestB) * 0.5f;
                EntityManager.TryGetComponent(edges[0], out Game.Net.Curve curve0);
                float d0 = math.distance(LaneMidpoint(curve0.m_Bezier).xz, midAB.xz);
                m_SelfTestEdge1 = d0 <= 15f ? edges[0] : edges[1];
                m_SelfTestEdge2 = d0 <= 15f ? edges[1] : edges[0];
                m_SelfTestPhase = 5;
            }
            else
            {
                m_SelfTestPhase = 15;
            }

            m_SelfTestWait = 4;
        }

        private void SelfTestPhaseStraighten()
        {
            // Bočni ofset krajeva OD čvora B — u njemu živi poravnanje traka;
            // ravnanje mora da ga SAČUVA (rotira), ne da ga obriše.
            SelfTestOffsetEndAt(m_SelfTestEdge1, 2f);
            SelfTestOffsetEndAt(m_SelfTestEdge2, -2f);

            ClearSelection();
            m_SelectedNodes.Add(m_SelfTestNodeB);
            bool straightened = StraightenSelectedNetNodes();
            SelfTestCheck(straightened, "RAVNANJE: ALT prihvatio čvor", string.Empty);
            m_SelfTestPhase = 6;
            m_SelfTestWait = 6;
        }

        private void SelfTestOffsetEndAt(Entity edge, float offset)
        {
            if (!EntityManager.TryGetComponent(edge, out Game.Net.Curve curve) ||
                !EntityManager.TryGetComponent(m_SelfTestNodeB, out Game.Net.Node nodeB))
            {
                return;
            }

            bool endAtB = math.distance(curve.m_Bezier.d.xz, nodeB.m_Position.xz) <=
                math.distance(curve.m_Bezier.a.xz, nodeB.m_Position.xz);
            if (endAtB)
            {
                curve.m_Bezier.d.z += offset;
            }
            else
            {
                curve.m_Bezier.a.z += offset;
            }

            float3 a = curve.m_Bezier.a;
            float3 d = curve.m_Bezier.d;
            curve.m_Bezier = new Bezier4x3(a, math.lerp(a, d, 1f / 3f), math.lerp(a, d, 2f / 3f), d);
            curve.m_Length = MathUtils.Length(curve.m_Bezier);
            EntityManager.SetComponentData(edge, curve);
            EntityManager.AddComponent<Updated>(edge);
        }

        private void SelfTestPhaseAssertStraighten()
        {
            bool haveNode = EntityManager.TryGetComponent(m_SelfTestNodeB, out Game.Net.Node nodeB);
            bool onLine = haveNode && math.abs(nodeB.m_Position.z - m_SelfTestA.z) < 1.5f;
            SelfTestCheck(onLine, "RAVNANJE: čvor legao na pravu A–C",
                haveNode ? string.Format(CultureInfo.InvariantCulture, "otklon {0:F2} m", math.abs(nodeB.m_Position.z - m_SelfTestA.z)) : "čvor mrtav");

            if (haveNode)
            {
                float offset1 = SelfTestEndOffset(m_SelfTestEdge1, nodeB.m_Position);
                float offset2 = SelfTestEndOffset(m_SelfTestEdge2, nodeB.m_Position);
                SelfTestCheck(offset1 > 1.2f && offset1 < 2.8f && offset2 > 1.2f && offset2 < 2.8f,
                    "RAVNANJE: bočni ofset (poravnanje traka) sačuvan",
                    string.Format(CultureInfo.InvariantCulture, "ofseti {0:F2} m i {1:F2} m (očekivano ~2)", offset1, offset2));
            }

            m_SelfTestPhase = 7;
            m_SelfTestWait = 2;
        }

        private float SelfTestEndOffset(Entity edge, float3 nodePosition)
        {
            if (!EntityManager.TryGetComponent(edge, out Game.Net.Curve curve))
            {
                return -1f;
            }

            float toStart = math.distance(curve.m_Bezier.a.xz, nodePosition.xz);
            float toEnd = math.distance(curve.m_Bezier.d.xz, nodePosition.xz);
            return math.min(toStart, toEnd);
        }

        private void SelfTestPhaseFilterAndHistory()
        {
            // Gašenje čipa mora da izbaci već selektovane deonice.
            ClearSelection();
            m_SelectedNetEdges.Add(m_SelfTestEdge1);
            m_SelectedNetEdges.Add(m_SelfTestEdge2);
            if (Mod.Settings != null)
            {
                Mod.Settings.SelectNetworks = false;
            }

            PurgeSelectionForDisabledFilters();
            SelfTestCheck(m_SelectedNetEdges.Count == 0 && m_SelectedNodes.Count == 0,
                "FILTER: gašenje čipa čisti selekciju", "ostalo " + m_SelectedNetEdges.Count);
            if (Mod.Settings != null)
            {
                Mod.Settings.SelectNetworks = true;
            }

            PurgeSelectionForDisabledFilters();

            // Učitavanje drugog grada mora da isprazni istoriju.
            SelfTestCheck(m_UndoStack.Count > 0, "ISTORIJA: undo stek pun pre pražnjenja", "koraka " + m_UndoStack.Count);
            DiscardWorldBoundState();
            SelfTestCheck(m_UndoStack.Count == 0 && m_RedoStack.Count == 0,
                "ISTORIJA: pražnjenje pri učitavanju grada", string.Empty);

            m_SelfTestPhase = 8;
            m_SelfTestWait = 12;
        }

        private readonly Dictionary<int, string> m_TrafficLabels = new Dictionary<int, string>();
        private readonly List<Entity> m_TrafficNodeLanes = new List<Entity>();
        private readonly List<Entity> m_TrafficEdge1Lanes = new List<Entity>();

        private static string SelfTestLaneKind(EntityManager em, Entity lane)
        {
            if (em.HasComponent<Game.Net.CarLane>(lane)) { return "car"; }
            if (em.HasComponent<Game.Net.PedestrianLane>(lane)) { return "ped"; }
            if (em.HasComponent<Game.Net.TrackLane>(lane)) { return "track"; }
            if (em.HasComponent<Game.Net.ConnectionLane>(lane)) { return "conn"; }
            if (em.HasComponent<Game.Net.ParkingLane>(lane)) { return "park"; }
            if (em.HasComponent<Game.Net.UtilityLane>(lane)) { return "util"; }
            return "?";
        }

        private void SelfTestCollectLanes(Entity owner, List<Entity> into)
        {
            into.Clear();
            if (EntityManager.TryGetBuffer(owner, true, out DynamicBuffer<Game.Net.SubLane> lanes))
            {
                for (int i = 0; i < lanes.Length; i++)
                {
                    into.Add(lanes[i].m_SubLane);
                }
            }
        }

        // Popis svega sto u ovom trenutku postoji u testnom skupu — po INDEKSU
        // entiteta, jer PathNode nosi samo indeks vlasnika.
        private void SelfTestBuildLabels()
        {
            m_TrafficLabels.Clear();
            m_TrafficLabels[m_SelfTestNodeB.Index] = "cvorB";
            m_TrafficLabels[m_SelfTestEdge1.Index] = "ivica1";
            m_TrafficLabels[m_SelfTestEdge2.Index] = "ivica2";

            foreach (Entity owner in new[] { m_SelfTestNodeB, m_SelfTestEdge1, m_SelfTestEdge2 })
            {
                if (!EntityManager.TryGetBuffer(owner, true, out DynamicBuffer<Game.Net.SubLane> lanes))
                {
                    continue;
                }

                string prefix = owner == m_SelfTestNodeB ? "cvorB" : (owner == m_SelfTestEdge1 ? "ivica1" : "ivica2");
                for (int i = 0; i < lanes.Length; i++)
                {
                    m_TrafficLabels[lanes[i].m_SubLane.Index] = prefix + ".traka" + i;
                }
            }
        }

        private string SelfTestOwnerLabel(int index)
        {
            return m_TrafficLabels.TryGetValue(index, out string label) ? label : ("e" + index + " ??? NEPOZNAT");
        }

        // Ispis svih traka jednog vlasnika + brojanje ustajalih referenci.
        private int SelfTestDumpOwner(string title, Entity owner)
        {
            if (!EntityManager.TryGetBuffer(owner, true, out DynamicBuffer<Game.Net.SubLane> lanes))
            {
                m_SelfTestReport.AppendLine("    " + title + ": nema SubLane bafera");
                return 0;
            }

            int stale = 0;
            m_SelfTestReport.AppendLine(string.Format(CultureInfo.InvariantCulture,
                "    {0} (e{1}) — {2} traka", title, owner.Index, lanes.Length));

            for (int i = 0; i < lanes.Length; i++)
            {
                Entity lane = lanes[i].m_SubLane;
                string kind = SelfTestLaneKind(EntityManager, lane);
                if (!EntityManager.TryGetComponent(lane, out Game.Net.Lane laneData))
                {
                    m_SelfTestReport.AppendLine("      #" + i + " e" + lane.Index + " " + kind + " — bez Lane komponente");
                    continue;
                }

                int startOwner = laneData.m_StartNode.GetOwnerIndex();
                int endOwner = laneData.m_EndNode.GetOwnerIndex();
                bool startBad = !m_TrafficLabels.ContainsKey(startOwner);
                bool endBad = !m_TrafficLabels.ContainsKey(endOwner);
                if (startBad || endBad)
                {
                    stale++;
                }

                m_SelfTestReport.AppendLine(string.Format(CultureInfo.InvariantCulture,
                    "      #{0} e{1} {2}  start={3}#{4}  end={5}#{6}{7}",
                    i, lane.Index, kind,
                    SelfTestOwnerLabel(startOwner), laneData.m_StartNode.GetLaneIndex(),
                    SelfTestOwnerLabel(endOwner), laneData.m_EndNode.GetLaneIndex(),
                    (startBad || endBad) ? "   << USTAJALO" : string.Empty));
            }

            return stale;
        }

        private void SelfTestDumpTraffic(string label)
        {
            SelfTestBuildLabels();
            m_SelfTestReport.AppendLine();
            m_SelfTestReport.AppendLine("  === " + label + " ===");

            if (EntityManager.TryGetBuffer(m_SelfTestNodeB, true, out DynamicBuffer<Game.Net.ConnectedEdge> connected))
            {
                m_SelfTestReport.AppendLine("    cvorB krakova: " + connected.Length);
            }

            // Kompozicija cvora zivi na IVICI (Composition.m_StartNode/m_EndNode
            // su prefab entiteti sa NetCompositionData) — otud se cita DeadEnd.
            if (EntityManager.TryGetComponent(m_SelfTestEdge1, out Game.Net.Composition composition) &&
                EntityManager.TryGetComponent(m_SelfTestEdge1, out Game.Net.Edge edge1Data))
            {
                Entity nodePrefab = edge1Data.m_End == m_SelfTestNodeB ? composition.m_EndNode : composition.m_StartNode;
                if (EntityManager.TryGetComponent(nodePrefab, out NetCompositionData compositionData))
                {
                    bool deadEnd = (compositionData.m_Flags.m_General & CompositionFlags.General.DeadEnd) != 0;
                    bool intersection = (compositionData.m_Flags.m_General & CompositionFlags.General.Intersection) != 0;
                    m_SelfTestReport.AppendLine("    kompozicija cvora: deadEnd=" + deadEnd + " intersection=" + intersection);
                    if (deadEnd)
                    {
                        m_SelfTestFailures++;
                        m_SelfTestReport.AppendLine("    !!! CVOR SA DVA KRAKA JE PROGLASEN SLEPOM ULICOM");
                    }
                }
                else
                {
                    m_SelfTestReport.AppendLine("    kompozicija cvora: nema NetCompositionData");
                }
            }
            else
            {
                m_SelfTestReport.AppendLine("    kompozicija: ivica1 nema Composition (jos nije preracunata)");
            }

            int stale = 0;
            stale += SelfTestDumpOwner("cvorB", m_SelfTestNodeB);
            stale += SelfTestDumpOwner("ivica1", m_SelfTestEdge1);
            stale += SelfTestDumpOwner("ivica2", m_SelfTestEdge2);

            m_SelfTestReport.AppendLine("    USTAJALIH REFERENCI: " + stale);
            if (stale > 0)
            {
                m_SelfTestFailures++;
            }
        }

        private void SelfTestPhaseTrafficBaseline()

        private static string ProbePath =>
            System.IO.Path.Combine(UnityEngine.Application.persistentDataPath, "ModsData", "Copaste", "probe.txt");

        private int m_ProbeCounter;

        // Automatski, bez prečice: posle svakog poteza nad mrezom. Kapa od 40
        // snimaka po sesiji da fajl ostane citljiv.
        partial void NetProbe(string reason)
        {
            if (m_ProbeCounter >= 40 || (m_SelectedNetEdges.Count == 0 && m_SelectedNodes.Count == 0))
            {
                return;
            }

            SelfTestProbeSelection(reason);

            // Zapamti pogodjene ivice i prati ih narednih 6 frejmova.
            m_FollowEdges.Clear();
            foreach (Entity edge in m_SelectedNetEdges)
            {
                m_FollowEdges.Add(edge);
            }

            foreach (Entity node in m_SelectedNodes)
            {
                if (EntityManager.TryGetBuffer(node, true, out DynamicBuffer<Game.Net.ConnectedEdge> connected))
                {
                    for (int i = 0; i < connected.Length; i++)
                    {
                        m_FollowEdges.Add(connected[i].m_Edge);
                    }
                }
            }

            m_FollowFrames = m_FollowEdges.Count > 0 ? 6 : 0;
            m_FollowLine.Clear();

            // Kasna provera: isti cvorovi ponovo za ~2 sekunde. Bez toga se ne
            // razlikuje trenutak usred preracunavanja od stvarno izgubljenih
            // traka — a bas to je pitanje.
            m_RecheckNodes.Clear();
            foreach (Entity node in m_SelectedNodes)
            {
                m_RecheckNodes.Add(node);
            }

            m_RecheckFrames = m_RecheckNodes.Count > 0 ? 120 : 0;
        }

        private readonly List<Entity> m_RecheckNodes = new List<Entity>();
        private int m_RecheckFrames;

        private void SelfTestRecheck()
        {
            if (m_RecheckFrames <= 0)
            {
                return;
            }

            if (--m_RecheckFrames > 0)
            {
                return;
            }

            try
            {
                StringBuilder text = new StringBuilder();
                text.AppendLine("    --- kasna provera (~2s kasnije) ---");
                foreach (Entity node in m_RecheckNodes)
                {
                    if (!EntityManager.Exists(node))
                    {
                        text.AppendLine("      cvor e" + node.Index + ": VISE NE POSTOJI");
                        continue;
                    }

                    int arms = EntityManager.TryGetBuffer(node, true, out DynamicBuffer<Game.Net.ConnectedEdge> connected)
                        ? connected.Length : -1;
                    int total = 0;
                    int cars = 0;
                    if (EntityManager.TryGetBuffer(node, true, out DynamicBuffer<Game.Net.SubLane> lanes))
                    {
                        total = lanes.Length;
                        for (int i = 0; i < lanes.Length; i++)
                        {
                            if (EntityManager.HasComponent<Game.Net.CarLane>(lanes[i].m_SubLane))
                            {
                                cars++;
                            }
                        }
                    }

                    bool broken = arms >= 2 && cars == 0;
                    float maxGap = 0f;
                    if (connected.IsCreated && EntityManager.TryGetComponent(node, out Game.Net.Node np))
                    {
                        for (int i = 0; i < connected.Length; i++)
                        {
                            if (EntityManager.TryGetComponent(connected[i].m_Edge, out Game.Net.Curve c) &&
                                EntityManager.TryGetComponent(connected[i].m_Edge, out Game.Net.Edge ed))
                            {
                                float3 end = ed.m_Start == node ? c.m_Bezier.a : c.m_Bezier.d;
                                maxGap = math.max(maxGap, math.length(end - np.m_Position));
                            }
                        }
                    }

                    text.AppendLine(string.Format(CultureInfo.InvariantCulture,
                        "      cvor e{0}: krakova={1} trake={2} voznih={3} najveci rascep={4:F2}m{5}{6}",
                        node.Index, arms, total, cars, maxGap,
                        broken ? "   <<< POKIDANA RASKRSNICA (dva kraka, nijedna vozna traka)" : string.Empty,
                        maxGap > 5f ? "   <<< PUT SE RAZDVOJIO" : string.Empty));
                }

                System.IO.File.AppendAllText(ProbePath, text.ToString());
            }
            catch (System.Exception)
            {
            }
        }

        private readonly List<Entity> m_FollowEdges = new List<Entity>();
        private readonly StringBuilder m_FollowLine = new StringBuilder();
        private int m_FollowFrames;

        // Po frejmu posle poteza: koliko traka pogodjenih ivica nosi Updated
        // odnosno PathfindUpdated. Nula kroz sve frejmove = pathfinding nije
        // ni saznao da se nesto promenilo.
        private void SelfTestFollowUp()
        {
            if (m_FollowFrames <= 0)
            {
                return;
            }

            m_FollowFrames--;

            int updated = 0;
            int pathfind = 0;
            int total = 0;
            foreach (Entity edge in m_FollowEdges)
            {
                if (!EntityManager.TryGetBuffer(edge, true, out DynamicBuffer<Game.Net.SubLane> lanes))
                {
                    continue;
                }

                for (int i = 0; i < lanes.Length; i++)
                {
                    total++;
                    if (EntityManager.HasComponent<Updated>(lanes[i].m_SubLane)) { updated++; }
                    if (EntityManager.HasComponent<Game.Common.PathfindUpdated>(lanes[i].m_SubLane)) { pathfind++; }
                }
            }

            // POUZDANA provera (ne zavisi od trenutka uzorkovanja): da li traka
            // jos uvek lezi na SVOJOJ deonici. Ako je deonica pomerena a traka
            // ostala na starom mestu, znaci da je LaneSystem nikad nije dirao —
            // vozila tada voze po geometriji koje vise nema.
            int stale = 0;
            float worst = 0f;
            foreach (Entity edge in m_FollowEdges)
            {
                if (!EntityManager.TryGetComponent(edge, out Game.Net.Curve edgeCurve) ||
                    !EntityManager.TryGetBuffer(edge, true, out DynamicBuffer<Game.Net.SubLane> lanes))
                {
                    continue;
                }

                for (int i = 0; i < lanes.Length; i++)
                {
                    if (!EntityManager.TryGetComponent(lanes[i].m_SubLane, out Game.Net.Curve laneCurve))
                    {
                        continue;
                    }

                    float d = math.min(
                        MathUtils.Distance(edgeCurve.m_Bezier.xz, laneCurve.m_Bezier.a.xz, out float _),
                        MathUtils.Distance(edgeCurve.m_Bezier.xz, laneCurve.m_Bezier.d.xz, out float _));
                    worst = math.max(worst, d);
                    if (d > 3f)
                    {
                        stale++;
                    }
                }
            }

            m_FollowLine.Append(string.Format(CultureInfo.InvariantCulture,
                " [{0}:{1}u/{2}p van={3} max={4:F1}m]", 6 - m_FollowFrames, updated, pathfind, stale, worst));

            if (m_FollowFrames == 0)
            {
                try
                {
                    string line = "    trake posle poteza (frejm:Updated/PathfindUpdated od " +
                        total + " ukupno):" + m_FollowLine + System.Environment.NewLine;
                    System.IO.File.AppendAllText(ProbePath, line);
                }
                catch (System.Exception)
                {
                }
            }
        }

        private void SelfTestProbeSelection(string reason)
        {
            try
            {

                m_ProbeCounter++;
                text.AppendLine();
                text.AppendLine("======== SNIMAK #" + m_ProbeCounter + " — " + reason + " ========");

            }

        private void ProbeNode(StringBuilder text, Entity node)
        {
            text.AppendLine("  --- CVOR e" + node.Index + " ---");
            if (EntityManager.TryGetComponent(node, out Game.Net.Node nodeData))
            {
                text.AppendLine(string.Format(CultureInfo.InvariantCulture,
                    "    pozicija: {0:F2} {1:F2} {2:F2}", nodeData.m_Position.x, nodeData.m_Position.y, nodeData.m_Position.z));
            }

            if (EntityManager.TryGetBuffer(node, true, out DynamicBuffer<Game.Net.ConnectedEdge> connected))
            {
                text.AppendLine("    krakova (ConnectedEdge): " + connected.Length);
                EntityManager.TryGetComponent(node, out Game.Net.Node nodePos);
                for (int i = 0; i < connected.Length; i++)
                {
                    Entity e = connected[i].m_Edge;
                    string mark = EntityManager.HasComponent<Deleted>(e) ? " DELETED" :
                        (EntityManager.HasComponent<Game.Tools.Temp>(e) ? " TEMP" : string.Empty);

                    // RASCEP: koliko je kraj krive te ivice udaljen od cvora.
                    // Kod autoputa mali ofset je normalan (poravnanje traka),
                    // ali metri znace da se put vizuelno pocepao.
                    string gap = string.Empty;
                    if (EntityManager.TryGetComponent(e, out Game.Net.Curve c) &&
                        EntityManager.TryGetComponent(e, out Game.Net.Edge ed))
                    {
                        float3 end = ed.m_Start == node ? c.m_Bezier.a : c.m_Bezier.d;
                        float3 d = end - nodePos.m_Position;
                        gap = string.Format(CultureInfo.InvariantCulture,
                            "  rascep={0:F2}m (xz={1:F2} visina={2:F2})",
                            math.length(d), math.length(d.xz), d.y);
                    }

                    text.AppendLine("      e" + e.Index + mark + gap);
                }
            }
            else
            {
                text.AppendLine("    NEMA ConnectedEdge bafer");
            }

            ProbeLanes(text, node, "cvor");
            ProbeNodeLaneDrift(text, node);
            ProbeFlags(text, node);

            // Rascep se ne mora videti na cvoru koji je pomeren nego na
            // SUPROTNOM kraju njegovih deonica — zato se i oni snimaju.
            if (EntityManager.TryGetBuffer(node, true, out DynamicBuffer<Game.Net.ConnectedEdge> arms))
            {
                for (int i = 0; i < arms.Length; i++)
                {
                    if (!EntityManager.TryGetComponent(arms[i].m_Edge, out Game.Net.Edge armEdge))
                    {
                        continue;
                    }

                    Entity far = armEdge.m_Start == node ? armEdge.m_End : armEdge.m_Start;
                    if (!EntityManager.Exists(far) || !EntityManager.TryGetComponent(far, out Game.Net.Node farData))
                    {
                        continue;
                    }

                    int farArms = EntityManager.TryGetBuffer(far, true, out DynamicBuffer<Game.Net.ConnectedEdge> fc) ? fc.Length : -1;
                    float farGap = 0f;
                    if (farArms > 0)
                    {
                        for (int k = 0; k < fc.Length; k++)
                        {
                            if (EntityManager.TryGetComponent(fc[k].m_Edge, out Game.Net.Curve fcv) &&
                                EntityManager.TryGetComponent(fc[k].m_Edge, out Game.Net.Edge fed))
                            {
                                float3 fend = fed.m_Start == far ? fcv.m_Bezier.a : fcv.m_Bezier.d;
                                farGap = math.max(farGap, math.length(fend - farData.m_Position));
                            }
                        }
                    }

                    int farCars = 0;
                    int farTotal = 0;
                    if (EntityManager.TryGetBuffer(far, true, out DynamicBuffer<Game.Net.SubLane> fl))
                    {
                        farTotal = fl.Length;
                        for (int k = 0; k < fl.Length; k++)
                        {
                            if (EntityManager.HasComponent<Game.Net.CarLane>(fl[k].m_SubLane))
                            {
                                farCars++;
                            }
                        }
                    }

                    text.AppendLine(string.Format(CultureInfo.InvariantCulture,
                        "    DALJI cvor e{0} (preko e{1}): krakova={2} trake={3} voznih={4} rascep={5:F2}m{6}{7}",
                        far.Index, arms[i].m_Edge.Index, farArms, farTotal, farCars, farGap,
                        (farArms >= 2 && farCars == 0) ? "   <<< POKIDANA RASKRSNICA" : string.Empty,
                        farGap > 3f ? "   <<< PUT SE RAZDVOJIO" : string.Empty));
                }
            }
        }

        // Da li trake CVORA prate cvor. Ako cvor pomerimo a trake ostanu na
        // starom mestu, ovo raste sa svakim pomeranjem — to je otisak toga da
        // ih igra nikad nije preracunala.
        private void ProbeNodeLaneDrift(StringBuilder text, Entity node)
        {
            if (!EntityManager.TryGetComponent(node, out Game.Net.Node nodeData) ||
                !EntityManager.TryGetBuffer(node, true, out DynamicBuffer<Game.Net.SubLane> lanes))
            {
                return;
            }

            float worst = 0f;
            for (int i = 0; i < lanes.Length; i++)
            {
                if (!EntityManager.TryGetComponent(lanes[i].m_SubLane, out Game.Net.Curve laneCurve))
                {
                    continue;
                }

                float d = math.min(
                    math.length(laneCurve.m_Bezier.a - nodeData.m_Position),
                    math.length(laneCurve.m_Bezier.d - nodeData.m_Position));
                worst = math.max(worst, d);
            }

            text.AppendLine(string.Format(CultureInfo.InvariantCulture,
                "    trake cvora najdalje od cvora: {0:F2}m{1}", worst,
                worst > 8f ? "   <<< TRAKE NISU PRATILE CVOR" : string.Empty));
        }

        private void ProbeEdge(StringBuilder text, Entity edge)
        {
            text.AppendLine("  --- DEONICA e" + edge.Index + " ---");
            if (EntityManager.TryGetComponent(edge, out Game.Net.Edge edgeData))
            {
                text.AppendLine("    krajevi: start=e" + edgeData.m_Start.Index + " end=e" + edgeData.m_End.Index);
            }

            if (EntityManager.TryGetComponent(edge, out Game.Net.Curve curve))
            {
                text.AppendLine(string.Format(CultureInfo.InvariantCulture,
                    "    duzina: {0:F2}", curve.m_Length));
            }

            if (EntityManager.TryGetComponent(edge, out Game.Net.Elevation elevation))
            {
                text.AppendLine(string.Format(CultureInfo.InvariantCulture,
                    "    elevacija: {0:F2} / {1:F2}", elevation.m_Elevation.x, elevation.m_Elevation.y));
            }
            else
            {
                text.AppendLine("    elevacija: nema komponente (prizemno)");
            }

            if (EntityManager.TryGetComponent(edge, out Game.Net.Aggregated aggregated))
            {
                text.AppendLine("    agregat (ime ulice): e" + aggregated.m_Aggregate.Index +
                    (EntityManager.Exists(aggregated.m_Aggregate) ? string.Empty : "  << NE POSTOJI"));
            }
            else
            {
                text.AppendLine("    agregat: NEMA  << deonica nije deo ulice");
            }

            // Zgrade zakacene na deonicu — bez njih nema ni vozila iz njih.
            if (EntityManager.TryGetBuffer(edge, true, out DynamicBuffer<Game.Buildings.ConnectedBuilding> buildings))
            {
                text.AppendLine("    zakacenih zgrada: " + buildings.Length);
            }

            // Vozila koja su trenutno na trakama ove deonice.
            ProbeLanes(text, edge, "deonica");
            ProbeFlags(text, edge);
        }

        private void ProbeLanes(StringBuilder text, Entity owner, string what)
        {
            if (!EntityManager.TryGetBuffer(owner, true, out DynamicBuffer<Game.Net.SubLane> lanes))
            {
                text.AppendLine("    NEMA SubLane bafer");
                return;
            }

            int cars = 0;
            int unsafeLanes = 0;
            int forbidden = 0;
            int laneObjects = 0;
            for (int i = 0; i < lanes.Length; i++)
            {
                Entity lane = lanes[i].m_SubLane;
                if (EntityManager.TryGetComponent(lane, out Game.Net.CarLane carLane))
                {
                    cars++;
                    if ((carLane.m_Flags & Game.Net.CarLaneFlags.Unsafe) != 0) { unsafeLanes++; }
                    if ((carLane.m_Flags & Game.Net.CarLaneFlags.Forbidden) != 0) { forbidden++; }
                }

                if (EntityManager.TryGetBuffer(lane, true, out DynamicBuffer<Game.Net.LaneObject> objects))
                {
                    laneObjects += objects.Length;
                }
            }

            text.AppendLine(string.Format(CultureInfo.InvariantCulture,
                "    trake: {0} ukupno, {1} voznih, {2} unsafe, {3} forbidden, {4} vozila na njima",
                lanes.Length, cars, unsafeLanes, forbidden, laneObjects));

            // Prve tri vozne trake sa vezama — da se vidi da li pokazuju na zivo.
            int shown = 0;
            for (int i = 0; i < lanes.Length && shown < 3; i++)
            {
                Entity lane = lanes[i].m_SubLane;
                if (!EntityManager.HasComponent<Game.Net.CarLane>(lane) ||
                    !EntityManager.TryGetComponent(lane, out Game.Net.Lane laneData))
                {
                    continue;
                }

                shown++;
                text.AppendLine(string.Format(CultureInfo.InvariantCulture,
                    "      vozna e{0}: start=e{1}#{2} end=e{3}#{4}",
                    lane.Index,
                    laneData.m_StartNode.GetOwnerIndex(), laneData.m_StartNode.GetLaneIndex(),
                    laneData.m_EndNode.GetOwnerIndex(), laneData.m_EndNode.GetLaneIndex()));
            }
        }

        private void ProbeFlags(StringBuilder text, Entity entity)
        {
            string flags = string.Empty;
            if (EntityManager.HasComponent<Updated>(entity)) { flags += " Updated"; }
            if (EntityManager.HasComponent<Game.Common.PathfindUpdated>(entity)) { flags += " PathfindUpdated"; }
            if (EntityManager.HasComponent<BatchesUpdated>(entity)) { flags += " BatchesUpdated"; }
            if (EntityManager.HasComponent<Game.Common.Created>(entity)) { flags += " Created"; }
            if (EntityManager.HasComponent<Deleted>(entity)) { flags += " Deleted"; }
            if (EntityManager.HasComponent<Game.Tools.Temp>(entity)) { flags += " Temp"; }
            if (EntityManager.HasComponent<Game.Net.Roundabout>(entity)) { flags += " Roundabout"; }
            if (EntityManager.HasComponent<Game.Net.TrafficLights>(entity)) { flags += " TrafficLights"; }
            text.AppendLine("    komponente:" + (flags.Length == 0 ? " (nijedna markerska)" : flags));
        }

        private void SelfTestPhaseCleanup()

            }

            m_SelfTestPhase = 16;
            m_SelfTestWait = 8;
        }

        private void SelfTestPhaseVerifyCleanup()
        {
            List<Entity> leftovers = SelfTestScan();
            SelfTestCheck(leftovers.Count == 0, "ČIŠĆENJE: ništa nije ostalo na mapi", "ostalo " + leftovers.Count);
            if (Mod.Settings != null)
            {
                Mod.Settings.SelectNetworks = m_SelfTestNetworksWas;
            }

            ClearSelection();

            // Vrati korisnikovu istoriju od pre testa (testovi su je ispraznili;
            // zapisi samog testa se NE vracaju — njihovi entiteti su pociscen).
            if (m_SelfTestUndoBackup != null)
            {
                m_UndoStack.Clear();
                m_UndoStack.AddRange(m_SelfTestUndoBackup);
                m_RedoStack.Clear();
                m_RedoStack.AddRange(m_SelfTestRedoBackup);
                m_SelfTestUndoBackup = null;
                m_SelfTestRedoBackup = null;
            }

            SelfTestFinish();
            m_SelfTestPhase = 0;
        }

        private void SelfTestFinish()
        {
            m_SelfTestReport.AppendLine();
            m_SelfTestReport.AppendLine(m_SelfTestFailures == 0 ? "SVE PROŠLO" : "PALO: " + m_SelfTestFailures);
            System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(SelfTestPath));
            System.IO.File.WriteAllText(SelfTestPath, m_SelfTestReport.ToString());
            Mod.Log.Info("Copaste self-test: " + (m_SelfTestFailures == 0 ? "SVE PROŠLO" : "PALO " + m_SelfTestFailures) + " — " + SelfTestPath);
            m_SelfTestReport = null;
        }
